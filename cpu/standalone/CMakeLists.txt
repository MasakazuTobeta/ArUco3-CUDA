# Build and validate the CPU experiment without a CUDA toolkit or GPU.
cmake_minimum_required(VERSION 3.24)
project(aruco3cpu_experiment VERSION 0.1.0 LANGUAGES CXX)
get_filename_component(PROJECT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${PROJECT_SOURCE_DIR}/cmake/Aruco3CudaOptions.cmake")
find_package(OpenCV 4.14 REQUIRED COMPONENTS core imgproc imgcodecs calib3d objdetect)
add_subdirectory("${PROJECT_SOURCE_DIR}/src/util" util)
add_subdirectory("${PROJECT_SOURCE_DIR}/reference" reference)
add_subdirectory("${PROJECT_SOURCE_DIR}/tools/corpusgen" corpusgen)
add_subdirectory("${PROJECT_SOURCE_DIR}/tools/cpu_quality" cpu_quality)
add_subdirectory("${PROJECT_SOURCE_DIR}/cpu" cpu)
if(ARUCO3CUDA_BUILD_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  include(GoogleTest)
  add_subdirectory("${PROJECT_SOURCE_DIR}/test/cpu" test)
endif()
aruco3cuda_add_format_targets()
aruco3cuda_add_coverage_target()
