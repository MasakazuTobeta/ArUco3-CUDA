cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(apriltag_nvidia_benchmark LANGUAGES CXX)
  set(kNvidiaDefault ON)
else()
  set(kNvidiaDefault OFF)
endif()
find_package(CUDAToolkit REQUIRED)
option(APRILTAG_BENCH_BUILD_NVIDIA "Link the external NVIDIA cuAprilTags benchmark" ${kNvidiaDefault})

if(APRILTAG_BENCH_BUILD_NVIDIA)
  find_path(CUAPRILTAGS_INCLUDE_DIR cuAprilTags.h REQUIRED)
  find_library(CUAPRILTAGS_LIBRARY cuapriltags REQUIRED)
  add_executable(apriltag_nvidia_benchmark nvidia_benchmark.cpp ../../examples/pgm.cpp)
  target_compile_features(apriltag_nvidia_benchmark PRIVATE cxx_std_17)
  target_include_directories(apriltag_nvidia_benchmark PRIVATE "${CUAPRILTAGS_INCLUDE_DIR}" ../../examples)
  target_link_libraries(apriltag_nvidia_benchmark PRIVATE "${CUAPRILTAGS_LIBRARY}" CUDA::cudart)
endif()
if(TARGET aruco3cuda_apriltag_gpu)
  add_executable(apriltag_project_benchmark project_benchmark.cpp ../../examples/pgm.cpp)
  target_compile_features(apriltag_project_benchmark PRIVATE cxx_std_17)
  target_include_directories(apriltag_project_benchmark PRIVATE ../../examples)
  target_link_libraries(apriltag_project_benchmark PRIVATE aruco3cuda::apriltag_gpu)
endif()

if(ARUCO3CUDA_BUILD_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME AprilTagBenchmarkProtocol
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_protocol.py")
endif()
